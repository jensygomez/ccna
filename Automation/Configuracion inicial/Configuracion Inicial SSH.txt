configure terminal

! Contraseña enable
enable secret cisco

! Crear usuario para SSH
username admin secret cisco

! Configuración de la consola
line console 0
 password cisco
 login
 logging synchronous
 exit

! Configuración de acceso remoto (VTY) para Telnet o SSH
line vty 0 4
 login local               ! Usará usuario "admin" para SSH
 transport input ssh telnet
 logging synchronous
 exit

! Configurar la interfaz VLAN1 con IP y activarla
interface vlan 1
 ip address 192.168.1.2 255.255.255.0
 no shutdown
 exit

! Configurar la puerta de enlace (default gateway) para acceder desde otra red
ip default-gateway 192.168.1.1

! Generar claves RSA para SSH
ip domain-name laboratorio.local
crypto key generate rsa
! Tamaño recomendado: 1024 o 2048 bits

end
write memory
